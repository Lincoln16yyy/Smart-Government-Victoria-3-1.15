# Smart Government Advisor - single always-present journal entry.
# The entry is only created while sg_enabled exists; disabling removes it
# through disable_evaluation, so the mod can be taken out of an existing save.

je_sg_law_advisor = {
	icon = "gfx/interface/icons/law_icons/autocracy.dds"
	group = je_group_internal_affairs

	activation_description = {
		custom_tooltip = sg_je_activation_tt
	}

	widget = {
		gui = "gui/journal_entry_widgets/sg_law_advisor.gui"
		name = "sg_law_advisor_widget"
		container = "custom_widget_container_3"
	}

	is_shown_when_inactive = {
		is_player = yes
		has_variable = sg_enabled
	}

	possible = {
		is_player = yes
		has_variable = sg_enabled
	}

	disable_evaluation = {
		NOT = { has_variable = sg_enabled }
	}

	can_deactivate = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = "sg_disabled_desc"
		trigger = {
			NOT = { has_variable = sg_enabled }
		}
	}

	# The 1.15 journal-entry pulse: no on_action file needs to be modified for
	# the periodic reassessment.
	on_monthly_pulse = {
		effect = {
			sg_monthly_update = yes
		}
	}
}
