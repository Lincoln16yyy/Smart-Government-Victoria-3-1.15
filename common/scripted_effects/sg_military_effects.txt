# The journal refresh and pulse both use this country-scope entry point.
sg_military_review = {
	if = {
		limit = { is_player = yes has_variable = sg_enabled }
		set_variable = { name = sg_military_army_size value = army_size }
		set_variable = { name = sg_military_navy_size value = navy_size }
		set_variable = { name = sg_military_balance value = net_total_income }
		sg_military_review_land_plan = yes
		set_variable = { name = sg_military_card value = 0 }
		if = {
			limit = { has_variable = sg_military_land_window }
			set_variable = { name = sg_military_card value = 1 }
		}
		else_if = {
			limit = {
				sg_military_land_eligible = yes
				NOT = { has_variable = sg_military_land_cooldown }
				any_scope_state = { sg_military_recruiting_state = yes }
				any_military_formation = { sg_military_recruiting_army = yes }
			}
			sg_military_open_land_plan = yes
		}
	}
}

sg_military_open_land_plan = {
	ordered_scope_state = {
		limit = { sg_military_recruiting_state = yes }
		order_by = population
		max = 1
		check_range_bounds = no
		save_scope_as = sg_military_site
	}
	every_military_formation = {
		limit = {
			sg_military_recruiting_army = yes
			owner = { NOT = { has_variable = sg_military_land_formation } }
		}
		save_scope_as = sg_military_formation
		owner = { set_variable = { name = sg_military_land_formation value = scope:sg_military_formation } }
	}
	set_variable = { name = sg_military_land_state value = scope:sg_military_site }
	set_variable = { name = sg_military_land_count value = { value = 10 subtract = army_size min = 1 max = 4 } }
	set_variable = { name = sg_military_land_goal value = { value = army_size add = var:sg_military_land_count } }
	set_variable = { name = sg_military_land_window days = 180 }
	remove_variable = sg_military_land_history
	set_variable = { name = sg_military_card value = 1 }
}

sg_military_review_land_plan = {
	if = {
		limit = { has_variable = sg_military_land_goal }
		if = {
			limit = { army_size >= var:sg_military_land_goal }
			sg_military_close_land_plan = yes
			set_variable = { name = sg_military_land_history value = 1 }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_variable = sg_military_land_window }
					sg_military_land_eligible = no
					NOT = { var:sg_military_land_state ?= { owner = root sg_military_recruiting_state = yes } }
					NOT = { var:sg_military_land_formation ?= { owner = root sg_military_recruiting_army = yes } }
				}
			}
			sg_military_close_land_plan = yes
			set_variable = { name = sg_military_land_history value = 2 }
		}
	}
}

sg_military_close_land_plan = {
	remove_variable = sg_military_land_count
	remove_variable = sg_military_land_goal
	remove_variable = sg_military_land_state
	remove_variable = sg_military_land_formation
	remove_variable = sg_military_land_window
	set_variable = { name = sg_military_land_cooldown days = 90 }
}

sg_military_clear = {
	sg_military_close_land_plan = yes
	remove_variable = sg_military_land_cooldown
	remove_variable = sg_military_land_history
	remove_variable = sg_military_army_size
	remove_variable = sg_military_navy_size
	remove_variable = sg_military_balance
	remove_variable = sg_military_card
}
