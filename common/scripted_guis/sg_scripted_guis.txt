# Smart Government Advisor - buttons used by the journal-entry widget.
# All of them are country-scope and only valid while sg_enabled exists.

sg_refresh_sgui = {
	scope = country
	is_shown = { is_player = yes has_variable = sg_enabled }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = {
		sg_rank_laws = yes
	}
}

sg_disable_sgui = {
	scope = country
	is_shown = { is_player = yes has_variable = sg_enabled }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = {
		sg_disable_country = yes
	}
}

sg_top_1_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_top_1_id var:sg_top_1_id > 0 }
}
sg_top_2_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_top_2_id var:sg_top_2_id > 0 }
}
sg_top_3_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_top_3_id var:sg_top_3_id > 0 }
}

sg_rank_diagnostic_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_last_rank_evaluated }
}

# Shown while any law is being enacted. During that period every generated
# availability trigger is false, so the journal must explain the pause instead
# of claiming that the recommended laws are already in force.
sg_enacting_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_enabled enacting_any_law = yes }
}

sg_top_1_opposition_visible_sgui = {

	is_shown = { has_variable = sg_top_1_opposition var:sg_top_1_opposition > 0 }
}
sg_top_2_opposition_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_top_2_opposition var:sg_top_2_opposition > 0 }
}
sg_top_3_opposition_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_top_3_opposition var:sg_top_3_opposition > 0 }
}

sg_ignored_1_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_ignored_1 }
}
sg_ignored_2_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_ignored_2 }

sg_ignored_3_visible_sgui = {
	scope = country
	is_shown = { has_variable = sg_ignored_3 }
}

# Enact and Ignore buttons are generated per law in sg_law_sguis.txt. Each
# button belongs to one law, so a rerank cannot retarget a click to another
# recommendation. The generated effect re-checks availability and
# enacting_any_law before starting normal enactment, then reranks.

sg_restore_ignored_1_sgui = {
	scope = country
	is_shown = { has_variable = sg_ignored_1 }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = { sg_restore_ignored_1 = yes }
}
sg_restore_ignored_2_sgui = {
	scope = country
	is_shown = { has_variable = sg_ignored_2 }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = { sg_restore_ignored_2 = yes }
}
sg_restore_ignored_3_sgui = {
	scope = country
	is_shown = { has_variable = sg_ignored_3 }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = { sg_restore_ignored_3 = yes }
}

sg_clear_ignored_sgui = {
	scope = country
	is_shown = { sg_has_ignored = yes }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = { sg_clear_ignored = yes }
}

sg_diagnostics_sgui = {
	scope = country

	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = { sg_log_diagnostics = yes }
}
