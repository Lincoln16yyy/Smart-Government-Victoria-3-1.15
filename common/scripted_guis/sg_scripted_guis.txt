# Smart Government Advisor - buttons used by the journal-entry widget.
# All of them are country-scope and only valid while sg_enabled exists.

sg_refresh_sgui = {
	scope = country
	is_shown = { is_player = yes has_variable = sg_enabled }
	is_valid = { is_player = yes has_variable = sg_enabled }
	effect = {
		sg_rank_laws = yes
	}
}
